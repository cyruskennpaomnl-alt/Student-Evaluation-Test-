// ─── Minigame: Catch the Blessings ───

let gameScore = 0;

const targetEmojis = ['☕','🕊️','⭐','📖','🤍','✨'];

let gameInterval;


function startGame() {

  gameScore = 0;

  document.getElementById('scoreText').textContent = gameScore;

  document.getElementById('gameArea').innerHTML = '';

  document.getElementById('unlockBtn').style.display = 'none';

  

  // Spawn a new item every 700ms

  gameInterval = setInterval(spawnGameTarget, 700);

}


function spawnGameTarget() {

  if (gameScore >= 6) return; 

  

  const area = document.getElementById('gameArea');

  const target = document.createElement('div');

  target.className = 'game-target';

  target.textContent = targetEmojis[Math.floor(Math.random() * targetEmojis.length)];

  

  // Random horizontal position

  const x = 10 + Math.random() * (area.clientWidth - 50);

  target.style.left = x + 'px';

  

  // Randomize speed

  const speed = 2.5 + Math.random() * 2; 

  target.style.animationDuration = `${speed}s, 1.5s`;


  // Click to catch

  target.onmousedown = target.ontouchstart = function(e) {

    e.preventDefault(); // Prevent double firing on touch

    if(target.classList.contains('pop-anim')) return;

    

    target.classList.add('pop-anim');

    target.textContent = '✨';

    

    gameScore++;

    document.getElementById('scoreText').textContent = gameScore;

    

    if(gameScore >= 6) {

      clearInterval(gameInterval);

      setTimeout(() => {

        area.innerHTML = '<div style="display:flex; height:100%; align-items:center; justify-content:center; color:var(--pink2); font-weight:800; font-size:1.4rem; animation:cardIn 0.5s;">🔓 Letter Unlocked! 🎉</div>';

        document.getElementById('unlockBtn').style.display = 'inline-flex';

      }, 500);

    }

  };


  area.appendChild(target);

  // Cleanup after animation

  setTimeout(() => { if(target.parentNode) target.remove(); }, speed * 1000);

}

function moveNoBtn(){

  const maxY = window.innerHeight - btn.offsetHeight - safeZone;

  const y = safeZone + Math.random() * (maxY - safeZone);

  

  btn.style.left = x + 'px';

  btn.style.top = y + 'px';

}

// ─── Confetti Explosion ───

function launchConfetti(){

  const colors = ['#ffb3c6','#ff85a1','#d4b8e0','#ffd6b3','#f9c846','#fff'];

  for(let i=0; i<150; i++){

    setTimeout(()=>{

      const c = document.createElement('div');

      c.style.position = 'fixed';

      c.style.zIndex = '200';

      c.style.pointerEvents = 'none';

      

      const size = 5 + Math.random()*8;

      const isCircle = Math.random() > 0.5;

      

      Object.assign(c.style,{

        left: Math.random()*100 + 'vw', top: '-20px',

        width: size+'px', height: size + (isCircle?0:size*0.5) +'px',

        borderRadius: isCircle ? '50%' : '4px',

        background: colors[Math.floor(Math.random()*colors.length)],

        transition: 'transform 3s cubic-bezier(0.25, 1, 0.5, 1), opacity 3s',

      });

      document.body.appendChild(c);

      

      // Trigger animation frame

      requestAnimationFrame(() => {

        c.style.transform = `translateY(${window.innerHeight + 50}px) rotate(${Math.random()*720}deg)`;

        c.style.opacity = '0';

      });

      

      setTimeout(()=>c.remove(), 3000);

    }, i*15);

  }

}
